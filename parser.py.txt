import json
import re

# Simulated IBM watsonx / IBM Bob Scope 3 Emission Parsing Model
def parse_invoice_and_calculate_emissions(file_path):
    with open(file_path, 'r') as file:
        content = file.read()

    # Regex patterns to extract invoice details
    vendor_match = re.search(r"Vendor:\s*(.*)", content)
    weight_match = re.search(r"Weight:\s*([\d\.]+)\s*tons", content)
    distance_match = re.search(r"Distance:\s*([\d\.]+)\s*km", content)

    vendor = vendor_match.group(1) if vendor_match else "Unknown"
    weight_tons = float(weight_match.group(1)) if weight_match else 0.0
    distance_km = float(distance_match.group(1)) if distance_match else 0.0

    # Emission Factor for Freight Transport: ~0.105 kg CO2e per ton-km
    EMISSION_FACTOR = 0.105
    ton_kilometers = weight_tons * distance_km
    total_emissions_kg = ton_kilometers * EMISSION_FACTOR

    output_data = {
        "project": "GreenWave AI",
        "vendor": vendor,
        "weight_tons": weight_tons,
        "distance_km": distance_km,
        "ton_km": ton_kilometers,
        "scope_3_emissions_kg_co2e": round(total_emissions_kg, 2)
    }

    return json.dumps(output_data, indent=4)

if __name__ == "__main__":
    result = parse_invoice_and_calculate_emissions("invoice.txt")
    print("--- Scope 3 Carbon Emission Extraction Output ---")
    print(result)
